6: import torch
7: from executorch.backends.qualcomm.builders.node_visitor import dq_ops, q_ops
8: from executorch.backends.qualcomm.builders.utils import is_parameter
9: from executorch.backends.qualcomm.utils.constants import QCOM_BYPASS_NODE
10: from executorch.exir.dialects._ops import ops as exir_ops
11: from executorch.exir.pass_base import ExportPass, PassResult
12: from executorch.exir.passes import dead_code_elimination_pass
13: 
14: 
15: class FoldQDQ(ExportPass):
16:     """
17:     Erase QDQ pattern.
18:     """
19: 
20:     def __init__(self, edge_program: torch.export.ExportedProgram, force_fold=False):
21:         super(FoldQDQ, self).__init__()
22:         self.edge_program = edge_program
23:         self.force_fold = force_fold
24: 
25:     def _annotate_bypass(self, node):
26:         node.meta[QCOM_BYPASS_NODE] = True
27:         for arg in node.args:
28:             if isinstance(arg, torch.fx.Node) and arg.op == "call_function":
29:                 self._annotate_bypass(arg)
30: 
31:     def _fold_dq(self, graph_module: torch.fx.GraphModule) -> torch.fx.GraphModule:
32:         # remove dq
33:         for n in graph_module.graph.nodes:
34:             user_list = list(n.users.keys())
35:             if n.target not in dq_ops:
36:                 continue
37: 
38:             # skip parameters & buffers
39:             if not self.force_fold and is_parameter(n.args[0], self.edge_program):
40:                 self._annotate_bypass(n)
41:             else:
42:                 for user_n in user_list:
43:                     user_n.replace_input_with(n, n.args[0])
44:                 graph_module.graph.erase_node(n)
45: 
46:     def _fold_q(self, graph_module: torch.fx.GraphModule) -> torch.fx.GraphModule:
47:         # remove q
48:         for n in graph_module.graph.nodes:
49:             if n.target not in q_ops:
50:                 continue
51: 
52:             to_be_removed = [n]
53:             source_n = n.args[0]
54: 
55:             # TODO: remove this hack as source_fn_stack is internal implementation detail of torch.export.
56:             # To make constant value/tensor be tagged as delegatable during partition
57:             if source_n.op == "get_attr":
58:                 source_n.meta["source_fn_stack"] = list(n.users.keys())[0].meta.get(
59:                     "source_fn_stack"
60:                 )
61: 
62:             # collecting quant nodes to be removed
63:             for i in range(1, len(n.args)):
64:                 if isinstance(n.args[i], torch.fx.node.Node):
65:                     to_be_removed.append(n.args[i])
66:                     # could be a commonly shared attribute between q & dq
67:                     if n.args[i].target == exir_ops.edge.aten._to_copy.default:
68:                         to_be_removed.append(n.args[i].args[0])
69:             # connect source node to quant users and remove quant node
70:             for user_n in list(n.users.keys()):
71:                 user_n.replace_input_with(n, n.args[0])
72:             for n in to_be_removed:
73:                 graph_module.graph.erase_node(n)
74: 
75:     def call(self, graph_module: torch.fx.GraphModule):
76:         self._fold_dq(graph_module)
77:         self._fold_q(graph_module)
78:         graph_module.recompile()
79:         dead_code_elimination_pass(graph_module)
80:         return PassResult(graph_module, True)
