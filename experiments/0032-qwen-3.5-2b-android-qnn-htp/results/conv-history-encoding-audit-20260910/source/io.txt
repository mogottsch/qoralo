120:     def _insert(self, graph_module: torch.fx.GraphModule) -> torch.fx.GraphModule:
121:         for n in graph_module.graph.nodes:
122:             # do nothing when a node is expected to output a quant tensor
123:             if n.meta.get(QCOM_QUANTIZED_IO):
124:                 continue
125: 
126:             # insert q after input or fold mix_quantization dq if applicable
127:             if (
128:                 n.op == "placeholder"
129:                 and n.meta.get(QCOM_QUANT_ATTRS)
130:                 and (
131:                     not is_parameter(n, self.edge_program)
132:                     or is_mutable_buffer_input(n, self.edge_program)
133:                 )
134:             ):
135:                 self._insert_quant_node(
136:                     graph_module, n, n.meta[QCOM_QUANT_ATTRS][QCOM_ENCODING]
137:                 )
138: 
139:             # insert dq before output or fold mix_quantization q if applicable
140:             users = list(n.users.keys())
141:             if n.meta.get(QCOM_QUANT_ATTRS) and any(
142:                 user.op == "output" for user in users
143:             ):
144:                 self._insert_dequant_node(
145:                     graph_module,
146:                     n,
147:                     self.q_dq_map[n.meta[QCOM_QUANT_ATTRS][QCOM_ENCODING]],
148:                 )
